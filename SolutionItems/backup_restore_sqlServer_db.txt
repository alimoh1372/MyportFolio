    INIT,

RESTORE DATABASE [PortfolioManager]
FROM DISK = N'pathOnDestinationServer\PortfolioManager_Full.bak'
WITH
    MOVE N'PortfolioManager'
        TO N'D:\SQLData\PortfolioManager.mdf',
    MOVE N'PortfolioDb_log'
        TO N'D:\SQLData\PortfolioManager_log_log.ldf',
    RECOVERY,
    CHECKSUM,
    STATS = 10;
